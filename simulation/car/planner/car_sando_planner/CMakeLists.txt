cmake_minimum_required(VERSION 3.16)
project(car_sando_planner)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-O2)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED)
find_package(OpenMP REQUIRED)
find_package(rclcpp REQUIRED)
find_package(dynus_interfaces REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(decomp_util REQUIRED)
find_package(decomp_ros_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(message_filters REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(pcl_ros REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(tf2_eigen REQUIRED)
find_package(std_srvs REQUIRED)
find_package(nlohmann_json REQUIRED)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}")
find_package(GUROBI REQUIRED)
file(GLOB CAR_GUROBI_SO "$ENV{GUROBI_HOME}/lib/libgurobi*[0-9].so")
if(NOT CAR_GUROBI_SO)
  message(FATAL_ERROR "Set GUROBI_HOME to an installed Linux Gurobi distribution")
endif()
include_directories(include ${EIGEN3_INCLUDE_DIR} ${PCL_INCLUDE_DIRS} ${DECOMP_UTIL_INCLUDE_DIRS} ${GUROBI_INCLUDE_DIRS})
link_directories(${PCL_LIBRARY_DIRS})
add_definitions(${PCL_DEFINITIONS})
add_executable(sando src/sando/sando_node.cpp src/sando/sando.cpp src/hgp/hgp_manager.cpp src/hgp/utils.cpp src/hgp/hgp_planner.cpp src/hgp/graph_search.cpp src/sando/utils.cpp src/sando/gurobi_solver.cpp)
ament_target_dependencies(sando rclcpp dynus_interfaces visualization_msgs decomp_util decomp_ros_msgs sensor_msgs std_msgs geometry_msgs nav_msgs message_filters pcl_conversions pcl_ros tf2 tf2_ros tf2_geometry_msgs tf2_eigen std_srvs nlohmann_json)
target_link_libraries(sando "$ENV{GUROBI_HOME}/lib/libgurobi_c++.a" ${CAR_GUROBI_SO} ${PCL_LIBRARIES} OpenMP::OpenMP_CXX Eigen3::Eigen nlohmann_json::nlohmann_json)
install(TARGETS sando DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
install(FILES LICENSE source_manifest.json DESTINATION share/${PROJECT_NAME})
ament_package()
